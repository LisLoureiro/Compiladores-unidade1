program exemplo;
var x, y, contador: integer;
    media, taxa: real;
begin
  x := 13;
  y := 4;
  contador := 0;
  media := 24.40e-04;
  taxa := 1.33;

  while contador < 10 do
  begin
    x := x + 1;
    contador := contador + 1;
  end;

  if x >= y and not (media < 2.5) then
    media := (x mod y) * 1.5
  else
    media := x / y + taxa;

  read(x);
  write(media);
end.
